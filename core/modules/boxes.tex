\usepackage{tcolorbox}
\tcbuselibrary{skins, breakable}

\tcbset{
\tcbset{
  locBase/.style={
    colback=white,
    colframe=black,
  },
}

    locNoteStyle/.style={
        locBase,
        colback=locInfoLight,
        colframe=locInfo,
        leftrule=5pt,
    },

    locExampleStyle/.style={
        locBase,
        colback=locPrimaryLight,
        colframe=locPrimary,
        toprule=3pt,
        bottomrule=3pt,
    },
}

\newtcolorbox{locNote}{locNoteStyle}
\newtcolorbox{locExample}{locExampleStyle}
