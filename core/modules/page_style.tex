% Header / Footer (KOMA)

\usepackage[headsepline]{scrlayer-scrpage}


% -------------------------------------------------
% Package extension hooks
% -------------------------------------------------

\providecommand{\locHeaderRightAddon}{}
\providecommand{\locFooterRightAddon}{}

\clearpairofpagestyles
\automark[section]{chapter}

\setkomafont{pageheadfoot}{\normalfont\sffamily\small\color{locTextMuted}}
\setkomafont{pagenumber}{\normalfont\sffamily\small\color{locTextMuted}}
\addtokomafont{headsepline}{\color{locLine}}

% -------------------------------------------------
% Headings
% -------------------------------------------------

\setkomafont{disposition}{\normalfont\sffamily\bfseries\color{locText}}
\setkomafont{chapter}{\normalfont\sffamily\huge\bfseries\color{locText}}
\setkomafont{section}{\normalfont\sffamily\Large\bfseries\color{locAccent}}
\setkomafont{subsection}{\normalfont\sffamily\large\bfseries\color{locText}}

\ihead{\textbf{\color{locAccent}loclass}}
\chead{}
\ohead{\headmark\hspace{1em}\locHeaderRightAddon}

\ifoot{}
\cfoot{}
\ofoot{\pagemark\hspace{1em}\locFooterRightAddon}

\setlength{\headheight}{18pt}


% -------------------------------------------------
% Title-page style
% -------------------------------------------------

\newpairofpagestyles{locclasstitle}{%
    \clearpairofpagestyles
    \ohead{\locHeaderRightAddon}
    \ofoot{\locFooterRightAddon}
}
